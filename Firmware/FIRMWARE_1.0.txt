#define F_CPU 8000000UL 							// CPU clock frequency with prescaler 1 (8 MHz)
#include <avr/io.h>
#include <util/delay.h>

	// ADC threshold (8-bit, VCC reference): 33/255 x 5 V = about 0.647 - 0.659 V at the PB2 pin
#define V_THRESHOLD 33

void delay_ms_31kHz(uint16_t ms) {
	while (ms--) {
		__builtin_avr_delay_cycles(31);       				// 1 ms = ~31 clock cycles @ 31.25 kHz
	}
}

void delay_ms_8MHz(uint16_t ms) {
	while (ms--) {
		__builtin_avr_delay_cycles(8000); 				// 1 ms = 8000 clock cycles @ 8 MHz
	}
}

										// CLOCK MANAGEMENT (Configuration Change Protection)

void set_clock_31kHz(void) {
	CCP = 0xD8; 								// Unlock register protection
	CLKPSR = (1 << CLKPS3); 						// Prescaler: 256 -> 8 MHz / 256 = 31.25 kHz
}

void set_clock_8MHz(void) {
	CCP = 0xD8; 								// Unlock register protection
	CLKPSR = 0x00; 								// Prescaler: 1 -> 8 MHz
}

// ADC measurement (8-bit)

uint8_t read_adc(void) {
	uint8_t result;
	
	ADCSRA |= (1 << ADEN); 							// Turning the ADC on
	ADCSRA |= (1 << ADSC); 							// Starting the measurement
	
	while (ADCSRA & (1 << ADSC)); 						// Waiting for the end of the measurement
	
	result = ADCL;
	ADCSRA &= ~(1 << ADEN); 						// Turning the ADC off
	
	return result;
}

int main(void) {
										// Declare variables at the very beginning of the block (required by the C89 standard)
	uint8_t v_meas;
	uint8_t i;

										// PIN PB0 set as the signal output
	DDRB |= (1 << PB0);
	PORTB &= ~(1 << PB0);
	
										// PIN PB2 set as the ADC input
	DDRB &= ~(1 << PB2);
	PUEB &= ~(1 << PB2);

										// (PB3 is the reset pin; its pull-up setting has an effect only if the reset function is disabled by the RSTDISBL fuse)
	PUEB |= (1 << PB1) | (1 << PB3);

										// ADC setup (VCC reference, PB2 channel, /64 prescaler)
	ADMUX = 2;
	ADCSRA = (1 << ADPS2) | (1 << ADPS1) | (0 << ADPS0);

	while (1) {
		phase_1:
										// 1. BOOT / RECOVERY PHASE
		set_clock_31kHz();
		delay_ms_31kHz(35); 						// 35 ms stabilization time for the MCP1640 and the capacitors

		phase_2:
										// 2. WAKE-UP AND MEASUREMENT PHASE
		set_clock_8MHz();
		
		v_meas = read_adc();

		if (v_meas < V_THRESHOLD) {
										// Voltage has dropped, switching to low-power standby monitoring
			set_clock_31kHz();
			while (1) {
				delay_ms_31kHz(10); 				// 10 ms rest at low clock frequency
				
				set_clock_8MHz(); 				// Wake up for the measurement
				if (read_adc() >= V_THRESHOLD) {
										// If the voltage has recovered, a mandatory recovery time follows
					goto phase_1;
				}
				set_clock_31kHz(); 				// If the voltage is still low, go back to the low-power standby
			}
		}

		// 3. PRECISION BURST PHASE
		// Timer0 Fast PWM (Mode 14), TOP = ICR0
		// Clock: 8 MHz, prescaler: 1 -> 1 tick = 0.125 us
		// Period (63 us) = 63 / 0.125 = 504 ticks -> ICR0 = 503
		// ON time (27.5 us) = 27.5 / 0.125 = 220 ticks -> OCR0A = 219
		
		// PWM Burst
		ICR0 = 503; 							// Full period: (503 + 1) x 0.125 us = 63 us
		OCR0A = 219; 							// ON time: (219 + 1) x 0.125 us = 27.5 us
		TIFR0 |= (1 << TOV0); 						// Clear any earlier interrupt/overflow flag (a flag is cleared by writing 1 to it)
		TCCR0A = (1 << COM0A1) | (1 << WGM01);  			// Set the output (PB0 / OC0A) to PWM mode: clear on compare match, set at BOTTOM
		TCCR0B = (1 << WGM03) | (1 << WGM02) | (1 << CS00); 		// Start the timer with prescaler 1

		// Let the CPU count 7 hardware-timed cycles
		for (i = 0; i < 7; i++) {
			while (!(TIFR0 & (1 << TOV0))); 			// Wait for the Timer0 overflow
			TIFR0 |= (1 << TOV0); 					// Clear the flag immediately for the next cycle
		}

		// Stop the burst immediately
		TCCR0B = 0;
		TCCR0A = 0;
		PORTB &= ~(1 << PB0); 						// Force a solid 0 V on the MOSFET gate driver input (keeps the MOSFET off)

		// 4. REST/WAITING PHASE
		set_clock_31kHz();
		delay_ms_31kHz(100); 						// 100 ms (nominal) passive wait with minimal power consumption
		// Start the next normal cycle
		goto phase_2;
	}
}
